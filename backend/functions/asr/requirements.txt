# Runtime dependencies for the asr Lambda.
#
# boto3 is already on the Lambda base image but we pin a known-good range
# here so local testing matches production.
#
# moonshine-onnx + numpy are only required when ASR_MODEL=moonshine (or
# moonshine-african). The default stub engine has no Python dependencies
# beyond the standard library and boto3.

boto3>=1.34,<2

# Real engine: Moonshine ONNX (Useful Sensors, MIT). The ONNX build is used
# instead of the Torch build to keep the Lambda container small and fast on
# CPU. Comment out to ship a stub-only image.
#
# VERSIONING: this package uses CalVer (YYYYMMDD), not SemVer — the published
# versions are 20251120, 20251121, ... A SemVer-style range like `>=0.2,<1`
# matches nothing and fails the image build outright, so pin a floor by date.
# Bump deliberately; there is no compatibility promise between dates.
useful-moonshine-onnx>=20251120
numpy>=1.26,<3
